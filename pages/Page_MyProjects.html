มีผู้ขอดำเนินการเบิกจ่ายโดยใช้เงินจากโครงการที่คุณเป็นเจ้าของ กรุณาตรวจสอบและกดส่งต่อเพื่อให้งานแผนงานดำเนินการต่อครับ

รหัสคำขอ / วันที่ ผู้ขอเบิก & รายการสิ่งของ ยอดขอเบิก จัดการ

รหัสโครงการ: ${projectId}
ปัญหาและอุปสรรค:

`, showCancelButton: true, confirmButtonText: 'สร้างเอกสารรายงาน', cancelButtonText: 'ยกเลิก', confirmButtonColor: '#8e44ad', customClass: { container: 'kanit-font' }, preConfirm: () => { let p = document.getElementById('report-problems').value.trim(); let s = document.getElementById('report-suggestions').value.trim(); if(!p) p = "-"; if(!s) s = "-"; return { problems: p, suggestions: s }; } }).then((result) => { if (result.isConfirmed) { Swal.fire({ title: 'กำลังสร้างเอกสารรายงาน...', html: 'ระบบกำลังดึงข้อมูลและสร้าง Google Docs
